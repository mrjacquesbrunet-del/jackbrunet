/* =====================================================================
   UN SEUL NOM - LE JAVASCRIPT DU SITE

   Un seul fichier pour les trois pages. Il se colle dans l'onglet
   JavaScript de l'extension " CSS et JS personnalises ".

   Il ne contient pas une seule esperluette, et ce n'est pas un hasard :
   WordPress les remplace par leur entite HTML, ce qui arreterait le
   navigateur sur une erreur de syntaxe. L'operateur " et " logique est
   donc ecrit " a ? b : false " d'un bout a l'autre du fichier.

   Les trois scripts sont independants et commencent chacun par une
   garde : s'il ne trouve pas la page qu'il sert, il s'arrete aussitot
   et ne touche a rien. Le fichier peut donc etre injecte sur toutes les
   pages du site sans risque.

   GENERE PAR generer-fichiers-wordpress.py - ne pas modifier a la main.
   ===================================================================== */



/* =====================================================================
   Page accueil (accueil.html)
   ===================================================================== */

/* Tout le code attend que la page soit construite. Sans cela, si le script
   est insere avant le contenu (ce que font certains plugins WordPress), il
   ne trouve aucun element et s'arrete sans rien faire. */
(function () {
  /* ---------------------------------------------------------------
     POURQUOI AUCUNE ESPERLUETTE DANS CE FICHIER

     WordPress retouche la typographie de tout ce qu'on lui confie. Sur
     du texte c'est utile ; sur du code c'est destructeur : il remplace
     l'esperluette par son entite HTML, et le navigateur s'arrete alors
     sur une erreur de syntaxe. Tout le script meurt, pas seulement la
     ligne. Constate en ligne le 20 octobre 2026 dans le bloc HTML de la
     page, puis de nouveau le 26 dans l'onglet JavaScript de
     l'extension : aucun des deux endroits n'est a l'abri.

     L'operateur " et " logique, qui s'ecrit avec deux esperluettes, est
     donc partout remplace par un ternaire " a ? b : false ". Il dit la
     meme chose et s'arrete de la meme facon quand a est faux, ce qui
     protege les acces du genre " a ? a.b : null ".

     Si tu ajoutes du code ici, n'y mets aucune esperluette : le
     generateur refuse de produire les fichiers tant qu'il en reste une.
     --------------------------------------------------------------- */
  function usnInit() {

    /* Le script est injecte par l'extension " CSS et JS personnalises ",
       donc sur TOUTES les pages du site. Celui-ci ne concerne que la
       page d'accueil : s'il ne la trouve pas, il s'arrete ici et ne
       touche a rien. */
    if (!document.getElementById('usn-accueil')) return;

  /* ---- Le menu du haut ----
     Le bouton a deux traits ouvre le panneau ; les deux traits se
     croisent, la page se bloque derriere, et Echap referme. */
  (function () {
    var bouton = document.getElementById('usn-bouton-menu');
    var panneau = document.getElementById('usn-menu');
    if (!bouton || !panneau) return;

    function basculer(ouvrir) {
      bouton.setAttribute('aria-expanded', ouvrir ? 'true' : 'false');
      bouton.setAttribute('aria-label', ouvrir ? 'Fermer le menu' : 'Ouvrir le menu');
      if (ouvrir) {
        panneau.hidden = false;
        // Un temps de rendu avant la classe, sinon la transition est sautee.
        requestAnimationFrame(function () { panneau.classList.add('est-ouvert'); });
      } else {
        panneau.classList.remove('est-ouvert');
        setTimeout(function () {
          if (bouton.getAttribute('aria-expanded') === 'false') panneau.hidden = true;
        }, 340);
      }
      document.documentElement.style.overflow = ouvrir ? 'hidden' : '';
    }

    bouton.addEventListener('click', function () {
      basculer(bouton.getAttribute('aria-expanded') !== 'true');
    });
    // Un lien clique referme le panneau : sinon l'ancre saute derriere lui.
    panneau.addEventListener('click', function (e) {
      if (e.target.closest('a')) basculer(false);
    });
    document.addEventListener('keydown', function (e) {
      if (e.key === 'Escape' ? bouton.getAttribute('aria-expanded') === 'true' : false) basculer(false);
    });
  })();

  /* ---- La loupe ----
     Un clic sur une affiche, qu'elle soit posee sur la carte ou en
     vignette, l'ouvre en grand. La meme fenetre sert a toutes. */
  (function () {
    var loupe  = document.getElementById('usn-loupe');
    var image  = document.getElementById('usn-loupe-img');
    var nom    = document.getElementById('usn-loupe-nom');
    var fermer = document.getElementById('usn-loupe-fermer');
    if (!loupe || !image) return;

    var rendre = null;   // a qui rendre le clavier en refermant

    function ouvrir(src, titre, origine) {
      image.src = src;
      image.alt = titre ? ('Affiche Un Seul Nom ' + titre) : '';
      if (nom) nom.textContent = titre || '';
      loupe.hidden = false;
      requestAnimationFrame(function () { loupe.classList.add('est-ouverte'); });
      document.documentElement.style.overflow = 'hidden';
      rendre = origine;
      if (fermer) fermer.focus();
    }

    function refermer() {
      loupe.classList.remove('est-ouverte');
      document.documentElement.style.overflow = '';
      setTimeout(function () {
        if (!loupe.classList.contains('est-ouverte')) { loupe.hidden = true; image.src = ''; }
      }, 280);
      if (rendre ? rendre.focus : false) rendre.focus();
      rendre = null;
    }

    var root = document.getElementById('usn-accueil') || document;
    root.addEventListener('click', function (e) {
      var cible = e.target.closest('[data-usn-affiche]');
      if (!cible) return;
      e.preventDefault();
      ouvrir(cible.getAttribute('data-usn-affiche'),
             cible.getAttribute('data-usn-titre') || '', cible);
    });

    // Sur la carte, les villes sont des groupes SVG : sans cela, on ne
    // peut pas les atteindre au clavier.
    var groupes = root.querySelectorAll('[data-usn-affiche]');
    for (var i = 0; i < groupes.length; i++) {
      groupes[i].setAttribute('tabindex', '0');
      groupes[i].setAttribute('role', 'button');
      groupes[i].addEventListener('keydown', function (e) {
        if (e.key === 'Enter' || e.key === ' ') {
          e.preventDefault();
          ouvrir(this.getAttribute('data-usn-affiche'),
                 this.getAttribute('data-usn-titre') || '', this);
        }
      });
    }

    loupe.addEventListener('click', function (e) {
      if (e.target === loupe || e.target === fermer) refermer();
    });
    document.addEventListener('keydown', function (e) {
      if (e.key === 'Escape' ? loupe.classList.contains('est-ouverte') : false) refermer();
    });
  })();

  /* ---- Le son de la bande-annonce ----
     Aucun navigateur ne lance seul une video qui parle : si on essaie,
     soit il refuse de la lancer du tout, soit il la force en silence.
     Elle demarre donc muette, et le visiteur reprend la main d'un clic
     n'importe ou sur l'image.

     Au premier clic, la bande-annonce repart de zero : elle raconte
     quelque chose, l'entendre a partir de la quarantieme seconde n'a
     pas de sens. */
  (function () {
    var video  = document.getElementById('usn-ouv-video');
    var bouton = document.getElementById('usn-ouv-son');
    var bande  = video ? video.closest('.usn-ouv-bande') : null;
    if (!video || !bouton) return;

    function aDuSon() {
      // Chaque navigateur renseigne un indicateur different, et aucun
      // n'est garanti : au moindre doute on garde le bouton.
      if (typeof video.mozHasAudio === 'boolean') return video.mozHasAudio;
      if (typeof video.webkitAudioDecodedByteCount === 'number') {
        return video.webkitAudioDecodedByteCount > 0;
      }
      if (video.audioTracks) return video.audioTracks.length > 0;
      return true;
    }

    video.addEventListener('loadeddata', function () {
      if (!aDuSon()) {
        bouton.remove();
        if (bande) bande.style.cursor = '';
      }
    });

    var jamaisTouche = true;
    bouton.classList.add('est-neuf');

    function basculer() {
      var muet = !video.muted;
      video.muted = muet;

      if (!muet) {
        if (jamaisTouche) { try { video.currentTime = 0; } catch (e) {} }
        var p = video.play();
        if (p ? p.catch : false) p.catch(function () {});
      }
      jamaisTouche = false;
      bouton.classList.remove('est-neuf');

      bouton.setAttribute('aria-pressed', muet ? 'false' : 'true');
      bouton.querySelector('span:last-child').textContent = muet ? 'Activer le son' : 'Couper le son';
      bouton.querySelector('span:first-child').innerHTML = muet ? '\u266A' : '\u2715';
    }

    bouton.addEventListener('click', function (e) { e.stopPropagation(); basculer(); });
    // Tout le cadre est cliquable : personne ne vise un petit bouton
    // quand l'envie lui prend d'entendre.
    if (bande) bande.addEventListener('click', basculer);
  })();

  /* ---- La barre se pose sur un fond quand on quitte le heros, et
          s'inverse quand elle passe au-dessus du bloc sombre ---- */
  (function () {
    var nav = document.getElementById('usn-nav');
    if (!nav) return;
    var sombres = document.querySelectorAll('#usn-accueil [data-usn-sombre]');
    var nom = nav.querySelector('.usn-nav-nom');

    function verifier() {
      nav.classList.toggle('est-posee', (window.pageYOffset || 0) > 40);

      // Vrai des qu'un bloc sombre passe derriere la barre.
      var sombre = false;
      for (var i = 0; i < sombres.length; i++) {
        var r = sombres[i].getBoundingClientRect();
        if (r.top <= 72 ? r.bottom > 72 : false) { sombre = true; break; }
      }
      nav.classList.toggle('est-nuit', sombre);
      if (nom) {
        nom.classList.toggle('usn-mot-blanc', sombre);
        nom.classList.toggle('usn-mot-noir', !sombre);
      }
    }
    window.addEventListener('scroll', verifier, { passive: true });
    window.addEventListener('resize', verifier);
    verifier();
  })();

  /* ---- La fenetre qui suit le doigt ou la souris ----
     Une fois la lame passee, c'est au visiteur de decouvrir le reste de
     la photo. On ecoute "pointermove" : le meme code sert donc a la
     souris, au doigt et au stylet. */
  (function () {
    var bloc   = document.getElementById('usn-bloc1');
    var hublot = document.getElementById('usn-hublot');
    var anneau = document.getElementById('usn-anneau');
    if (!bloc || !hublot) return;

    var vues = hublot.querySelectorAll('img');
    var iVue = 0, dernierEchange = 0;
    var cible = { x: 50, y: 46 }, courant = { x: 50, y: 46 };
    var actif = false, anime = false;

    function rayon() {
      // 17vmin, la meme valeur que le CSS, mais en pixels : l'anneau doit
      // tomber exactement sur le bord du masque.
      return Math.min(window.innerWidth, window.innerHeight) * 0.17;
    }

    function allumer(oui) {
      if (actif === oui) return;
      actif = oui;
      hublot.classList.toggle('est-actif', oui);
      if (anneau) anneau.classList.toggle('est-actif', oui);
    }

    function boucle(t) {
      courant.x += (cible.x - courant.x) * 0.14;
      courant.y += (cible.y - courant.y) * 0.14;
      bloc.style.setProperty('--mx', courant.x.toFixed(2) + '%');
      bloc.style.setProperty('--my', courant.y.toFixed(2) + '%');

      if (anneau) {
        var r = rayon();
        anneau.style.setProperty('--usn-rayon', r.toFixed(1) + 'px');
        anneau.style.transform = 'translate3d('
          + (courant.x / 100 * bloc.clientWidth - r).toFixed(1) + 'px,'
          + (courant.y / 100 * bloc.clientHeight - r).toFixed(1) + 'px,0)';
      }

      // On change de photo au bout d'un moment, pas a chaque mouvement :
      // sinon les images clignotent des qu'on bouge la souris.
      var tourner = vues.length > 1 ? (actif ? t - dernierEchange > 1800 : false) : false;
      if (tourner) {
        dernierEchange = t;
        vues[iVue].classList.remove('est-vue');
        iVue = (iVue + 1) % vues.length;
        vues[iVue].classList.add('est-vue');
      }

      var bouge = Math.abs(cible.x - courant.x) > 0.05 || Math.abs(cible.y - courant.y) > 0.05;
      if (bouge || actif) { requestAnimationFrame(boucle); } else { anime = false; }
    }

    function lancer() { if (!anime) { anime = true; requestAnimationFrame(boucle); } }

    // Au doigt, un geste vertical veut dire "fais defiler la page", pas
    // "ouvre la fenetre". On ne suit donc le doigt que s'il part de cote :
    // sans cela, chaque defilement declenchait le repeint d'une photo
    // plein ecran et la page devenait poussive.
    var ancre = null;

    function suivre(e) {
      if (e.pointerType === 'touch') {
        if (!ancre) return;
        var dx = Math.abs(e.clientX - ancre.x), dy = Math.abs(e.clientY - ancre.y);
        if (dy > dx) return;
        if (dx < 8) return;
      }
      var r = bloc.getBoundingClientRect();
      if (!r.width) return;
      cible.x = ((e.clientX - r.left) / r.width) * 100;
      cible.y = ((e.clientY - r.top) / r.height) * 100;
      allumer(true);
      lancer();
    }

    // Passif : le doigt continue de faire defiler la page normalement.
    bloc.addEventListener('pointermove', suivre, { passive: true });
    bloc.addEventListener('pointerdown', function (e) {
      if (e.pointerType === 'touch') { ancre = { x: e.clientX, y: e.clientY }; return; }
      suivre(e);
    }, { passive: true });
    bloc.addEventListener('pointerleave', function () { allumer(false); });
    // Au telephone, le doigt quitte l'ecran : on laisse la fenetre
    // ouverte un instant plutot que de l'effacer d'un coup.
    function relacher() {
      ancre = null;
      setTimeout(function () { allumer(false); }, 1400);
    }
    bloc.addEventListener('pointerup', relacher, { passive: true });
    bloc.addEventListener('pointercancel', relacher, { passive: true });
  })();

  /* =================================================================
     MOTEUR D'ANIMATION AU SCROLL
     Reproduit le comportement de GSAP ScrollTrigger en mode " scrub " :
     - les elements partent d'une demi-hauteur d'ecran plus bas (50svh),
     - ils remontent AU RYTHME DU SCROLL (et redescendent si on remonte),
     - avec un chevauchement entre les elements d'un meme groupe,
     - le tout lisse (inertie) pour un rendu fluide.
     Sans JS ou avec " prefers-reduced-motion ", tout reste visible.
     ================================================================= */
  (function () {
    var root = document.getElementById('usn-accueil');
    if (!root) return;
    var reduce = window.matchMedia ? window.matchMedia('(prefers-reduced-motion: reduce)').matches : false;

    function clamp01(v) { return v < 0 ? 0 : v > 1 ? 1 : v; }
    function easeOut(t) { return 1 - Math.pow(1 - t, 3); }
    function vh() { return window.innerHeight || 800; }

    /* ---------- 1. Decoupe la punchline en mots ---------- */
    var punch = document.getElementById('usn-punch');
    var words = [];
    if (punch) {
      var lines = punch.querySelectorAll('[data-usn-reveal]');
      for (var l = 0; l < lines.length; l++) {
        var parts = (lines[l].textContent || '').trim().split(/\s+/);
        lines[l].textContent = '';
        for (var k = 0; k < parts.length; k++) {
          var sp = document.createElement('span');
          sp.className = 'usn-word';
          sp.textContent = parts[k];
          lines[l].appendChild(sp);
          if (k < parts.length - 1) lines[l].appendChild(document.createTextNode(' '));
          words.push(sp);
        }
      }
      for (var w = 0; w < words.length; w++) {
        var a = Math.sin(w * 12.9898) * 43758.5453; a -= Math.floor(a);
        var b = Math.sin(w * 78.233) * 12543.123;  b -= Math.floor(b);
        words[w].__dx = (a - 0.5) * 2 * Math.min(window.innerWidth * 0.45, 520);
        words[w].__dy = (b - 0.5) * 2 * 260;
        words[w].__rot = (a - 0.5) * 55;
      }
    }

    if (reduce) return; // mouvement desactive : tout reste en place

    /* ---------- 2. Declare les groupes animes ---------- */
    // rise : monte depuis 50svh ; slide : arrive de la gauche ; pop : grossit
    var groups = [];
    function group(sel, opts) {
      var els = root.querySelectorAll(sel);
      if (els.length) groups.push({ els: els, o: opts });
    }
    group('[data-usn-monte]',   { type: 'rise',  dist: 0.28, stagger: 0.10, span: 0.52 });
    group('.usn-vue',           { type: 'rise',  dist: 0.36, stagger: 0.07, span: 0.54 });
    group('.usn-chiffre',       { type: 'rise',  dist: 0.22, stagger: 0.10, span: 0.46 });
    group('[data-usn-gauche]',  { type: 'slide',  dist: 0.55, stagger: 0, span: 0.52 });
    group('[data-usn-droite]',  { type: 'slideR', dist: 0.55, stagger: 0, span: 0.52 });

    var all = [];
    for (var g = 0; g < groups.length; g++) {
      for (var i = 0; i < groups[g].els.length; i++) {
        var el = groups[g].els[i];
        all.push({
          el: el, i: i, o: groups[g].o, p: 0, top: 0,
          rot: parseFloat(el.getAttribute('data-rot') || '0')
        });
      }
    }

    function scrollY() { return window.pageYOffset || document.documentElement.scrollTop || 0; }

    /* Position NATURELLE de chaque element (transform neutralise), mise en
       cache : sinon le decalage applique fausse la mesure et l'element
       n'entre jamais dans la zone de declenchement. */
    function measureAll() {
      for (var i = 0; i < all.length; i++) {
        var it = all[i], prev = it.el.style.transform;
        it.el.style.transform = 'none';
        it.top = it.el.getBoundingClientRect().top + scrollY();
        it.el.style.transform = prev;
      }
    }
    measureAll();

    /* ---------- Portraits : couleur au passage au centre ----------
       Chaque photo se colore a mesure qu'elle approche du centre de
       l'ecran, puis retourne au noir et blanc en s'en eloignant. */
    var photos = [];
    (function () {
      var imgs = root.querySelectorAll('.usn-vue img, .usn-visage img');
      for (var i = 0; i < imgs.length; i++) photos.push({ el: imgs[i], p: 0 });
    })();

    function cibleCouleur(ph) {
      var r = ph.el.getBoundingClientRect();
      if (!r.height) return 0;
      var h = vh();
      var ecart = Math.abs((r.top + r.height / 2) - h / 2);
      // Pleine couleur au centre, noir et blanc au-dela de 42% de l'ecran.
      return clamp01(1 - ecart / (h * 0.42));
    }

    function peindre(ph) {
      var e = easeOut(ph.p);
      ph.el.style.filter = 'grayscale(' + (1 - e).toFixed(3) + ') contrast(' + (1 + 0.06 * (1 - e)).toFixed(3) + ')';
      ph.el.style.transform = 'scale(' + (1 + 0.035 * e).toFixed(4) + ')';
    }
    for (var ip = 0; ip < photos.length; ip++) peindre(photos[ip]);

    // etat initial immediat (evite tout " flash " de contenu deja en place)
    for (var s = 0; s < all.length; s++) apply(all[s], 0);

    /* ---------- 3. Calcul de l'avancement lie au scroll ---------- */
    function target(item) {
      var h = vh();
      var top = item.top - scrollY();
      // 0 quand le haut de l'element entre par le bas, 1 quand il a monte
      var raw = (h * 0.92 - top) / (h * item.o.span);
      var st = item.i * item.o.stagger;
      return clamp01((clamp01(raw) - st) / Math.max(0.2, 1 - st));
    }

    function apply(item, p) {
      var e = easeOut(p), inv = 1 - e, o = item.o, st = item.el.style;
      var rot = item.rot ? ' rotate(' + item.rot + 'deg)' : '';
      if (o.type === 'rise') {
        st.transform = 'translate3d(0,' + (inv * o.dist * vh()).toFixed(1) + 'px,0)' + rot;
      } else if (o.type === 'slide') {
        st.transform = 'translate3d(' + (-inv * o.dist * Math.min(window.innerWidth, 900)).toFixed(1) + 'px,0,0)' + rot;
      } else if (o.type === 'slideR') {
        st.transform = 'translate3d(' + (inv * o.dist * Math.min(window.innerWidth, 900)).toFixed(1) + 'px,0,0)' + rot;
      } else { // pop
        st.transform = 'scale(' + (0.3 + 0.7 * e).toFixed(3) + ')' + rot;
      }
      st.opacity = e.toFixed(3);
    }

    /* ---------- 4. Punchline : les mots se rassemblent ---------- */
    var punchP = 0;
    // Ancre sur la premiere ligne de texte (et non sur la section, bien plus
    // haute) : les mots se rassemblent des que le texte entre a l'ecran.
    var punchAnchor = punch ? punch.querySelector('[data-usn-reveal]') : null;
    function punchTarget() {
      if (!punchAnchor) return 0;
      var r = punchAnchor.getBoundingClientRect(), h = vh();
      return clamp01((h * 1.0 - r.top) / (h * 0.5));
    }
    function punchApply(p) {
      var n = words.length;
      for (var i = 0; i < n; i++) {
        var t = clamp01((p - (i / n) * 0.45) / 0.55);
        var e = easeOut(t), inv = 1 - e, ws = words[i];
        ws.style.transform = 'translate3d(' + (ws.__dx * inv).toFixed(1) + 'px,' + (ws.__dy * inv).toFixed(1) + 'px,0) rotate(' + (ws.__rot * inv).toFixed(1) + 'deg)';
        ws.style.opacity = e.toFixed(3);
        ws.style.filter = inv > 0.02 ? 'blur(' + (inv * 9).toFixed(1) + 'px)' : 'none';
      }
      var sub = punch ? punch.querySelector('[data-usn-sub]') : null;
      if (sub) {
        var sp = clamp01((p - 0.6) / 0.4);
        sub.style.opacity = sp.toFixed(3);
        sub.style.transform = 'translate3d(0,' + ((1 - sp) * 40).toFixed(1) + 'px,0)';
      }
    }
    punchApply(0);

    /* ---------- 5. La phrase ----------
       Les mots s'allument l'un apres l'autre. On ne parcourt pas toute la
       hauteur du bloc : la phrase est entierement lue aux trois quarts,
       ce qui laisse un temps de lecture avant de repartir. */
    var phrase = root.querySelector('.usn-phrase');
    var motsPh = phrase ? phrase.querySelectorAll('#usn-phrase-mots b') : [];

    function phraseAppliquer() {
      if (!phrase || !motsPh.length) return;
      var r = phrase.getBoundingClientRect();
      var c = Math.max(1, r.height - vh());
      var p = clamp01(clamp01(-r.top / c) / 0.75);
      var jusqua = p * motsPh.length;
      for (var i = 0; i < motsPh.length; i++) {
        var dit = i < jusqua;
        if (motsPh[i].__dit !== dit) {
          motsPh[i].__dit = dit;
          motsPh[i].classList.toggle('est-dit', dit);
        }
      }
    }
    phraseAppliquer();

    /* ---------- 5 bis. La frise ----------
       Le defilement vertical tire la bande vers la gauche. La hauteur du
       bloc est calee sur la largeur reelle de la bande : la traversee se
       fait donc a la meme vitesse sur un telephone et sur un grand ecran,
       au lieu de defiler en un eclair d'un cote et de trainer de l'autre. */
    var frise  = root.querySelector('.usn-frise');
    var piste  = document.getElementById('usn-frise-piste');
    var jauge  = document.getElementById('usn-frise-part');
    var signe  = document.getElementById('usn-fr-signe');
    var signes = signe ? signe.querySelectorAll('path') : [];
    var course = 0;

    for (var g = 0; g < signes.length; g++) {
      var lg2 = 0;
      try { lg2 = signes[g].getTotalLength(); } catch (e) { lg2 = 0; }
      signes[g].__lg = lg2;
      if (lg2) {
        signes[g].style.strokeDasharray = lg2;
        signes[g].style.strokeDashoffset = lg2;
      }
    }

    function mesurerFrise() {
      if (!frise || !piste) return;
      course = Math.max(0, piste.scrollWidth - window.innerWidth);
      // Un peu moins d'un pixel de defilement par pixel parcouru : la
      // bande avance sans obliger a une descente interminable.
      frise.style.height = (vh() + course * 0.85) + 'px';
    }

    function friseAppliquer() {
      if (!frise || !piste) return;
      var r = frise.getBoundingClientRect();
      var c = Math.max(1, r.height - vh());
      var p = clamp01(-r.top / c);
      piste.style.transform = 'translate3d(' + (-course * p).toFixed(1) + 'px,0,0)';
      if (jauge) jauge.style.width = (p * 100).toFixed(2) + '%';

      // La signature se trace quand son panneau traverse l'ecran.
      if (signe ? signes.length : 0) {
        var rs = signe.getBoundingClientRect();
        var q = clamp01((window.innerWidth * 0.78 - rs.left) / (window.innerWidth * 0.42));
        for (var i = 0; i < signes.length; i++) {
          var tr = signes[i];
          if (!tr.__lg) continue;
          var qi = clamp01((q - i * 0.05) / 0.45);
          tr.style.strokeDashoffset = (tr.__lg * (1 - qi)).toFixed(1);
        }
      }
    }
    mesurerFrise();
    friseAppliquer();

    /* ---------- 6. Parallaxes ----------
       Chaque element marque monte a sa propre vitesse pendant qu'il
       traverse l'ecran : c'est ce qui donne de la profondeur a la
       colonne d'affiches. */
    var paras = root.querySelectorAll('[data-usn-parallaxe]');
    function parallaxes() {
      for (var i = 0; i < paras.length; i++) {
        var el = paras[i], r = el.getBoundingClientRect(), h = vh();
        var p = clamp01((h - r.top) / (h + r.height));
        var v = parseFloat(el.getAttribute('data-usn-parallaxe')) || 1;
        el.style.transform = 'translate3d(0,' + ((0.5 - p) * 150 * v).toFixed(1) + 'px,0)';
      }
    }
    parallaxes();

    /* ---------- 7. Les deux visages qui se rapprochent ---------- */
    var visages = root.querySelector('.usn-visages');
    function visagesAppliquer() {
      if (!visages) return;
      var r = visages.getBoundingClientRect(), h = vh();
      // 0 quand le bloc entre par le bas, 1 quand il est au centre
      var p = clamp01((h - r.top) / (h * 0.85 + r.height * 0.4));
      var e = easeOut(p), ecart = (1 - e) * Math.min(window.innerWidth * 0.42, 460);
      var g = visages.querySelector('[data-usn-visage="gauche"]');
      var d = visages.querySelector('[data-usn-visage="droite"]');
      if (g) g.style.transform = 'translate3d(' + (-ecart).toFixed(1) + 'px,0,0)';
      if (d) d.style.transform = 'translate3d(' + ecart.toFixed(1) + 'px,0,0)';
    }
    visagesAppliquer();

    var hero = null, heroInner = null;

    /* ---------- 6. Boucle de rendu (lissage type ScrollSmoother) ---------- */
    var running = false;
    var dernierY = -1;
    function frame() {
      var moving = false;

      // Ces trois-la suivent le scroll directement : tant qu'il bouge,
      // la boucle continue.
      var y = scrollY();
      if (y !== dernierY) { dernierY = y; moving = true; }
      phraseAppliquer();
      friseAppliquer();
      parallaxes();
      visagesAppliquer();

      for (var i = 0; i < all.length; i++) {
        var it = all[i], t = target(it);
        it.p += (t - it.p) * 0.12;             // lissage
        if (Math.abs(t - it.p) > 0.001) moving = true; else it.p = t;
        apply(it, it.p);
      }

      if (punch) {
        var tp = punchTarget();
        punchP += (tp - punchP) * 0.12;
        if (Math.abs(tp - punchP) > 0.001) moving = true; else punchP = tp;
        punchApply(punchP);
      }

      for (var k = 0; k < photos.length; k++) {
        var ph = photos[k], tc = cibleCouleur(ph);
        ph.p += (tc - ph.p) * 0.14;
        if (Math.abs(tc - ph.p) > 0.004) moving = true; else ph.p = tc;
        peindre(ph);
      }

      if (hero) {
        var hr = hero.getBoundingClientRect();
        var prog = clamp01(-hr.top / Math.max(1, hr.height));
        if (heroInner) {
          heroInner.style.transform = 'translate3d(0,' + (prog * 130).toFixed(1) + 'px,0)';
          heroInner.style.opacity = (1 - prog * 0.85).toFixed(3);
        }
      }

      if (moving) { requestAnimationFrame(frame); } else { running = false; }
    }
    function kick() { if (!running) { running = true; requestAnimationFrame(frame); } }

    window.addEventListener('scroll', kick, { passive: true });
    window.addEventListener('resize', function () { measureAll(); mesurerFrise(); kick(); });
    window.addEventListener('load', function () { measureAll(); mesurerFrise(); kick(); });
    kick();
  })();


  /* ---- Lettre d'information ---- */
  (function () {
    var form = document.getElementById('usn-accueil-form');
    if (!form) return;
    var msg = document.getElementById('usn-form-msg');

    form.addEventListener('submit', function (e) {
      var nom   = form.querySelector('#usn-nom');
      var email = form.querySelector('#usn-email');
      if (!nom.value.trim() || !/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(email.value)) {
        e.preventDefault();
        (nom.value.trim() ? email : nom).focus();
        return;
      }
      if (msg) {
        msg.textContent = 'Merci ! Vous serez pr\u00e9venu d\u00e8s qu\'une nouvelle date est annonc\u00e9e.';
        msg.style.display = 'block';
      }
      setTimeout(function () { form.reset(); }, 100);
    });
  })();

  }
  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', usnInit);
  } else {
    usnInit();
  }
})();

/* Tout le code attend que la page soit construite. Sans cela, si le script
   est insere avant le contenu (ce que font certains plugins WordPress), il
   ne trouve aucun element et s'arrete sans rien faire. */
(function () {
  /* ---------------------------------------------------------------
     POURQUOI AUCUNE ESPERLUETTE DANS CE FICHIER

     WordPress retouche la typographie de tout ce qu'on lui confie. Sur
     du texte c'est utile ; sur du code c'est destructeur : il remplace
     l'esperluette par son entite HTML, et le navigateur s'arrete alors
     sur une erreur de syntaxe. Tout le script meurt, pas seulement la
     ligne. Constate en ligne le 20 octobre 2026 dans le bloc HTML de la
     page, puis de nouveau le 26 dans l'onglet JavaScript de
     l'extension : aucun des deux endroits n'est a l'abri.

     L'operateur " et " logique, qui s'ecrit avec deux esperluettes, est
     donc partout remplace par un ternaire " a ? b : false ". Il dit la
     meme chose et s'arrete de la meme facon quand a est faux, ce qui
     protege les acces du genre " a ? a.b : null ".

     Si tu ajoutes du code ici, n'y mets aucune esperluette : le
     generateur refuse de produire les fichiers tant qu'il en reste une.
     --------------------------------------------------------------- */
  function usnInit() {

    /* Le script est injecte par l'extension " CSS et JS personnalises ",
       donc sur TOUTES les pages du site. Celui-ci ne concerne que la
       page de la ville : s'il ne la trouve pas, il s'arrete ici et ne
       touche a rien. */
    if (!document.getElementById('usn-event')) return;

  /* ---- Le grand titre s'ajuste toujours a la largeur ----
     " AVIGNON " debordait sur mobile ; avec un nom de ville plus long le
     probleme reviendrait. On mesure et on reduit si necessaire. */
  (function () {
    var h1 = document.querySelector('#usn-event .usn-titre h1');
    if (!h1) return;
    function fit() {
      h1.style.fontSize = '';                       // repart de la taille CSS
      var avail = h1.clientWidth;
      if (!avail) return;
      var size = parseFloat(getComputedStyle(h1).fontSize);
      for (var i = 0; i < 3 ? h1.scrollWidth > avail : false; i++) {
        size = Math.floor(size * (avail / h1.scrollWidth) * 0.99);
        h1.style.fontSize = size + 'px';
      }
    }
    fit();
    window.addEventListener('resize', fit);
    if (document.fonts ? document.fonts.ready : false) document.fonts.ready.then(fit);
  })();

  /* ---- Son de la video ----
     Aucun navigateur ne lance seul une video qui parle : elle demarre
     donc muette, et ce bouton rend la main au visiteur. S'il n'y a pas
     de piste audio dans le fichier, le bouton se retire de lui-meme
     plutot que de promettre un son qui n'existe pas. */
  (function () {
    var video  = document.querySelector('#usn-event .usn-hero-video');
    var bouton = document.getElementById('usn-son');
    if (!video || !bouton) return;

    function aDuSon() {

      // disponible partout, d'ou les trois essais successifs.
      if (typeof video.mozHasAudio === 'boolean') return video.mozHasAudio;
      if (video.webkitAudioDecodedByteCount !== undefined) {
        return video.webkitAudioDecodedByteCount > 0;
      }
      if (video.audioTracks) return video.audioTracks.length > 0;
      return true;   // dans le doute, on laisse le bouton
    }

    function verifier() {
      if (!aDuSon()) bouton.remove();
    }
    if (video.readyState >= 2) { verifier(); }
    video.addEventListener('loadeddata', verifier);

    function libelle() {
      var texte = bouton.querySelector('.usn-son-texte');
      var icone = bouton.querySelector('.usn-son-icone');
      if (texte) texte.textContent = video.muted ? 'Activer le son' : 'Couper le son';
      if (icone) icone.textContent = video.muted ? '\u266A' : '\u266B';
      bouton.setAttribute('aria-pressed', video.muted ? 'false' : 'true');
    }

      libelle();
    });
  })();

  /* ---- Compte a rebours ---- */
  (function () {
    var box = document.getElementById('usn-countdown');
    if (!box) return;
    var target = new Date(box.getAttribute('data-target') || '').getTime();
    if (isNaN(target)) return;
    var els = {
      days: box.querySelector('[data-cd="days"]'),
      hours: box.querySelector('[data-cd="hours"]'),
      mins: box.querySelector('[data-cd="mins"]'),
      secs: box.querySelector('[data-cd="secs"]')
    };
    function pad(n) { return (n < 10 ? '0' : '') + n; }
    function tick() {
      var diff = Math.max(0, target - Date.now());
      var s = Math.floor(diff / 1000);
      var d = Math.floor(s / 86400); s -= d * 86400;
      var h = Math.floor(s / 3600); s -= h * 3600;
      var m = Math.floor(s / 60); s -= m * 60;
      if (els.days) els.days.textContent = d;
      if (els.hours) els.hours.textContent = pad(h);
      if (els.mins) els.mins.textContent = pad(m);
      if (els.secs) els.secs.textContent = pad(s);
    }
    tick();
    setInterval(tick, 1000);
  })();

  /* =================================================================
     MOTEUR D'ANIMATION AU SCROLL
     Reproduit le comportement de GSAP ScrollTrigger en mode " scrub " :
     - les elements partent d'une demi-hauteur d'ecran plus bas (50svh),
     - ils remontent AU RYTHME DU SCROLL (et redescendent si on remonte),
     - avec un chevauchement entre les elements d'un meme groupe,
     - le tout lisse (inertie) pour un rendu fluide.
     Sans JS ou avec " prefers-reduced-motion ", tout reste visible.
     ================================================================= */
  (function () {
    var root = document.getElementById('usn-event');
    if (!root) return;
    var reduce = window.matchMedia ? window.matchMedia('(prefers-reduced-motion: reduce)').matches : false;

    function clamp01(v) { return v < 0 ? 0 : v > 1 ? 1 : v; }
    function easeOut(t) { return 1 - Math.pow(1 - t, 3); }
    function vh() { return window.innerHeight || 800; }

    /* ---------- 1. Decoupe la punchline en mots ---------- */
    var punch = document.getElementById('usn-punch');
    var words = [];
    if (punch) {
      var lines = punch.querySelectorAll('[data-usn-reveal]');
      for (var l = 0; l < lines.length; l++) {
        var parts = (lines[l].textContent || '').trim().split(/\s+/);
        lines[l].textContent = '';
        for (var k = 0; k < parts.length; k++) {
          var sp = document.createElement('span');
          sp.className = 'usn-word';
          sp.textContent = parts[k];
          lines[l].appendChild(sp);
          if (k < parts.length - 1) lines[l].appendChild(document.createTextNode(' '));
          words.push(sp);
        }
      }
      for (var w = 0; w < words.length; w++) {
        var a = Math.sin(w * 12.9898) * 43758.5453; a -= Math.floor(a);
        var b = Math.sin(w * 78.233) * 12543.123;  b -= Math.floor(b);
        words[w].__dx = (a - 0.5) * 2 * Math.min(window.innerWidth * 0.45, 520);
        words[w].__dy = (b - 0.5) * 2 * 260;
        words[w].__rot = (a - 0.5) * 55;
      }
    }

    if (reduce) return; // mouvement desactive : tout reste en place

    /* ---------- 2. Declare les groupes animes ---------- */
    // rise : monte depuis 50svh ; slide : arrive de la gauche ; pop : grossit
    var groups = [];
    function group(sel, opts) {
      var els = root.querySelectorAll(sel);
      if (els.length) groups.push({ els: els, o: opts });
    }
    // Les deux boutons du bandeau se rejoignent : l'un vient de la
    // gauche, l'autre de la droite. Le reste du bandeau - logo, ville,
    // accroche, date - et le compte a rebours restent immobiles.
    group('[data-usn-gauche]', { type: 'slide',  dist: 0.60, stagger: 0, span: 0.50 });
    group('[data-usn-droite]', { type: 'slideR', dist: 0.60, stagger: 0, span: 0.50 });
    group('.usn-prog',        { type: 'slide', dist: 0.42, stagger: 0.16, span: 0.62 });
    group('.usn-person',      { type: 'rise',  dist: 0.50, stagger: 0.13, span: 0.60 });
    group('.usn-datepill',    { type: 'pop',   stagger: 0,    span: 0.55 });
    group('.usn-cta-card',    { type: 'rise',  dist: 0.40, stagger: 0.20, span: 0.60 });
    group('.usn-date-row',    { type: 'slide', dist: 0.30, stagger: 0.16, span: 0.60 });
    group('[data-usn-side="left"]',  { type: 'slide',  dist: 0.55, stagger: 0.10, span: 0.85 });
    group('[data-usn-side="right"]', { type: 'slideR', dist: 0.55, stagger: 0.10, span: 0.85 });
    group('.usn-section-head',{ type: 'rise',  dist: 0.28, stagger: 0.10, span: 0.55 });
    group('.usn-fact',        { type: 'rise',  dist: 0.32, stagger: 0.12, span: 0.58 });

    var all = [];
    for (var g = 0; g < groups.length; g++) {
      for (var i = 0; i < groups[g].els.length; i++) {
        var el = groups[g].els[i];
        all.push({
          el: el, i: i, o: groups[g].o, p: 0, top: 0,
          rot: parseFloat(el.getAttribute('data-rot') || '0')
        });
      }
    }

    function scrollY() { return window.pageYOffset || document.documentElement.scrollTop || 0; }

    /* Position NATURELLE de chaque element (transform neutralise), mise en
       cache : sinon le decalage applique fausse la mesure et l'element
       n'entre jamais dans la zone de declenchement. */
    function measureAll() {
      for (var i = 0; i < all.length; i++) {
        var it = all[i], prev = it.el.style.transform;
        it.el.style.transform = 'none';
        it.top = it.el.getBoundingClientRect().top + scrollY();
        it.el.style.transform = prev;
      }
    }
    measureAll();

    /* ---------- Portraits : couleur au passage au centre ----------
       Chaque photo se colore a mesure qu'elle approche du centre de
       l'ecran, puis retourne au noir et blanc en s'en eloignant. */
    var photos = [];
    (function () {
      var imgs = root.querySelectorAll('.usn-person-photo img');
      for (var i = 0; i < imgs.length; i++) photos.push({ el: imgs[i], p: 0 });
    })();

    function cibleCouleur(ph) {
      var r = ph.el.getBoundingClientRect();
      if (!r.height) return 0;
      var h = vh();
      var ecart = Math.abs((r.top + r.height / 2) - h / 2);
      // Pleine couleur au centre, noir et blanc au-dela de 42% de l'ecran.
      return clamp01(1 - ecart / (h * 0.42));
    }

    function peindre(ph) {
      var e = easeOut(ph.p);
      ph.el.style.filter = 'grayscale(' + (1 - e).toFixed(3) + ') contrast(' + (1 + 0.06 * (1 - e)).toFixed(3) + ')';
      ph.el.style.transform = 'scale(' + (1 + 0.035 * e).toFixed(4) + ')';
    }
    for (var ip = 0; ip < photos.length; ip++) peindre(photos[ip]);

    // etat initial immediat (evite tout " flash " de contenu deja en place)
    for (var s = 0; s < all.length; s++) apply(all[s], 0);

    /* ---------- 3. Calcul de l'avancement lie au scroll ---------- */
    function target(item) {
      var h = vh();
      var top = item.top - scrollY();
      // 0 quand le haut de l'element entre par le bas, 1 quand il a monte
      var raw = (h * 0.92 - top) / (h * item.o.span);
      var st = item.i * item.o.stagger;
      return clamp01((clamp01(raw) - st) / Math.max(0.2, 1 - st));
    }

    function apply(item, p) {
      var e = easeOut(p), inv = 1 - e, o = item.o, st = item.el.style;
      var rot = item.rot ? ' rotate(' + item.rot + 'deg)' : '';
      if (o.type === 'rise') {
        st.transform = 'translate3d(0,' + (inv * o.dist * vh()).toFixed(1) + 'px,0)' + rot;
      } else if (o.type === 'slide') {
        st.transform = 'translate3d(' + (-inv * o.dist * Math.min(window.innerWidth, 900)).toFixed(1) + 'px,0,0)' + rot;
      } else if (o.type === 'slideR') {
        st.transform = 'translate3d(' + (inv * o.dist * Math.min(window.innerWidth, 900)).toFixed(1) + 'px,0,0)' + rot;
      } else { // pop
        st.transform = 'scale(' + (0.3 + 0.7 * e).toFixed(3) + ')' + rot;
      }
      st.opacity = e.toFixed(3);
    }

    /* ---------- 4. Punchline : les mots se rassemblent ---------- */
    var punchP = 0;
    // Ancre sur la premiere ligne de texte (et non sur la section, bien plus
    // haute) : les mots se rassemblent des que le texte entre a l'ecran.
    var punchAnchor = punch ? punch.querySelector('[data-usn-reveal]') : null;
    function punchTarget() {
      if (!punchAnchor) return 0;
      var r = punchAnchor.getBoundingClientRect(), h = vh();
      return clamp01((h * 1.0 - r.top) / (h * 0.5));
    }
    function punchApply(p) {
      var n = words.length;
      for (var i = 0; i < n; i++) {
        var t = clamp01((p - (i / n) * 0.45) / 0.55);
        var e = easeOut(t), inv = 1 - e, ws = words[i];
        ws.style.transform = 'translate3d(' + (ws.__dx * inv).toFixed(1) + 'px,' + (ws.__dy * inv).toFixed(1) + 'px,0) rotate(' + (ws.__rot * inv).toFixed(1) + 'deg)';
        ws.style.opacity = e.toFixed(3);
        ws.style.filter = inv > 0.02 ? 'blur(' + (inv * 9).toFixed(1) + 'px)' : 'none';
      }
      var sub = punch ? punch.querySelector('[data-usn-sub]') : null;
      if (sub) {
        var sp = clamp01((p - 0.6) / 0.4);
        sub.style.opacity = sp.toFixed(3);
        sub.style.transform = 'translate3d(0,' + ((1 - sp) * 40).toFixed(1) + 'px,0)';
      }
    }
    punchApply(0);

    /* ---------- 5. Parallaxe du heros ---------- */
    var hero = root.querySelector('.usn-hero');
    var heroInner = root.querySelector('.usn-hero-inner');

    /* ---------- 6. Boucle de rendu (lissage type ScrollSmoother) ---------- */
    var running = false;
    function frame() {
      var moving = false;

      for (var i = 0; i < all.length; i++) {
        var it = all[i], t = target(it);
        it.p += (t - it.p) * 0.12;             // lissage
        if (Math.abs(t - it.p) > 0.001) moving = true; else it.p = t;
        apply(it, it.p);
      }

      if (punch) {
        var tp = punchTarget();
        punchP += (tp - punchP) * 0.12;
        if (Math.abs(tp - punchP) > 0.001) moving = true; else punchP = tp;
        punchApply(punchP);
      }

      for (var k = 0; k < photos.length; k++) {
        var ph = photos[k], tc = cibleCouleur(ph);
        ph.p += (tc - ph.p) * 0.14;
        if (Math.abs(tc - ph.p) > 0.004) moving = true; else ph.p = tc;
        peindre(ph);
      }

      if (hero) {
        var hr = hero.getBoundingClientRect();
        var prog = clamp01(-hr.top / Math.max(1, hr.height));
        if (heroInner) {
          heroInner.style.transform = 'translate3d(0,' + (prog * 130).toFixed(1) + 'px,0)';
          heroInner.style.opacity = (1 - prog * 0.85).toFixed(3);
        }
      }

      if (moving) { requestAnimationFrame(frame); } else { running = false; }
    }
    function kick() { if (!running) { running = true; requestAnimationFrame(frame); } }

    window.addEventListener('scroll', kick, { passive: true });
    window.addEventListener('resize', function () { measureAll(); kick(); });
    window.addEventListener('load', function () { measureAll(); kick(); });
    kick();
  })();

  /* ---- Reservation des places + envoi vers Brevo ----
     Le comptage est fait par le serveur (module " Un Seul Nom - Reservations ").
     Si l'API n'est pas disponible, le compteur reste masque et le formulaire
     continue de fonctionner comme un simple envoi vers Brevo. */
  (function () {
    var form = document.getElementById('usn-signup-form');
    if (!form) return;

    var API      = '/wp-json/usn/v1/';
    var msg      = document.getElementById('usn-form-msg');
    var err      = document.getElementById('usn-form-err');
    var stock    = document.getElementById('usn-stock');
    var champPl  = document.getElementById('usn-champ-places');
    var choixPl  = document.getElementById('usn-places');

    function valeurPlaces() {
      var coche = choixPl ? choixPl.querySelector('input[name="PLACES"]:checked') : null;
      return coche ? parseInt(coche.value, 10) || 1 : 1;
    }
    var bouton   = form.querySelector('button[type=submit]');
    var apiActive = false;
    var maxPlaces = 1;

    function erreur(texte, genre) {
      if (!err) return;
      err.textContent = texte;
      err.style.display = texte ? 'block' : 'none';
      // " deja inscrit " s'affiche en bleu : c'est une bonne nouvelle
      // mal formulee en rouge.
      err.classList.toggle('est-info', 'info' === genre);
    }

    function afficherEtat(e) {
      var restantes = Math.max(0, e.restantes | 0);
      var capacite  = Math.max(1, e.capacite | 0);
      var complet   = restantes === 0 || e.ouvert === false;

      if (stock) {
        stock.classList.add('usn-on');
        stock.classList.toggle('usn-complet', complet);
        var num = stock.querySelector('[data-usn-restantes]');
        var lab = stock.querySelector('[data-usn-libelle]');
        var jau = stock.querySelector('[data-usn-jauge]');
        if (num) num.textContent = complet ? 'Complet' : restantes;
        if (lab) lab.textContent = complet ? '' : (restantes > 1 ? 'places restantes' : 'place restante');
        if (jau) jau.style.width = Math.min(100, ((capacite - restantes) / capacite) * 100) + '%';
      }

      // Pastilles : bornees par la limite et par ce qu'il reste reellement.
      maxPlaces = Math.max(1, Math.min(e.max | 0 || 1, restantes));
      if (choixPl ? champPl : null) {
        var garde = valeurPlaces();
        choixPl.innerHTML = '';
        for (var i = 1; i <= maxPlaces; i++) {
          var lab2 = document.createElement('label');
          lab2.className = 'usn-place';
          var input = document.createElement('input');
          input.type = 'radio';
          input.name = 'PLACES';
          input.value = i;
          if (i === (garde || 1)) input.checked = true;
          var sp = document.createElement('span');
          sp.appendChild(document.createTextNode(i));
          var pt = document.createElement('small');
          pt.textContent = i > 1 ? 'places' : 'place';
          sp.appendChild(pt);
          lab2.appendChild(input);
          lab2.appendChild(sp);
          choixPl.appendChild(lab2);
        }
        champPl.style.display = complet ? 'none' : '';

        var hint = champPl.querySelector('[data-usn-hint]');
        if (hint) {
          var limite = e.max | 0 || 1;
          hint.textContent = complet ? ''
            : (restantes < limite
                ? 'Il ne reste que ' + restantes + ' place(s) disponible(s).'
                : 'Limite de ' + limite + ' places par r\u00e9servation.');
        }
      }

      if (bouton) {
        bouton.disabled = complet;
        bouton.textContent = complet ? 'Complet' : "Je r\u00e9serve ma place";
      }
    }

    // Lecture de l'etat au chargement.
    if (window.fetch) {
      fetch(API + 'places', { credentials: 'same-origin' })
        .then(function (r) { return r.ok ? r.json() : Promise.reject(); })
        .then(function (e) { apiActive = true; afficherEtat(e); })
        .catch(function () { /* module absent : on garde le formulaire simple */ });
    }

    function envoyerVersBrevo() {
      // Le formulaire vise l'iframe cache : le visiteur reste sur la page.
      form.submit();
    }

    function succes(restantes) {
      if (msg) {
        msg.textContent = 'Merci ! Ta r\u00e9servation est enregistr\u00e9e. \u00c0 tr\u00e8s bient\u00f4t.';
        msg.style.display = 'block';
      }
      erreur('');
      setTimeout(function () { form.reset(); }, 100);
    }

    form.addEventListener('submit', function (e) {
      var nom   = form.querySelector('#usn-nom');
      var email = form.querySelector('#usn-email');
      if (!nom.value.trim() || !/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(email.value)) {
        e.preventDefault();
        (nom.value.trim() ? email : nom).focus();
        return;
      }

      // Sans module de reservation : comportement precedent (Brevo seul).
      if (!apiActive || !window.fetch) { succes(); return; }

      // Avec module : on reserve d'abord, on n'envoie a Brevo qu'en cas de succes.
      e.preventDefault();
      erreur('');
      if (bouton) { bouton.disabled = true; bouton.textContent = 'Reservation...'; }

      fetch(API + 'reserver', {
        method: 'POST',
        credentials: 'same-origin',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          nom: nom.value.trim(),
          email: email.value.trim(),
          places: valeurPlaces(),
          site: ''
        })
      })
        .then(function (r) { return r.json().then(function (d) { return { ok: r.ok, d: d }; }); })
        .then(function (res) {
          if (bouton) { bouton.disabled = false; bouton.textContent = 'Je r\u00e9serve ma place'; }
          var reussi = res.ok ? (res.d ? res.d.ok : false) : false;
          if (reussi) {
            envoyerVersBrevo();
            succes(res.d.restantes);
            fetch(API + 'places', { credentials: 'same-origin' })
              .then(function (r) { return r.json(); }).then(afficherEtat).catch(function () {});
          } else {
            var motif = res.d ? res.d.erreur : '';
            erreur((res.d ? res.d.message : '') || 'La r\u00e9servation a \u00e9chou\u00e9, r\u00e9essaie.',
                   'deja-inscrit' === motif ? 'info' : '');
            if (res.d ? typeof res.d.restantes === 'number' : false) {
              fetch(API + 'places', { credentials: 'same-origin' })
                .then(function (r) { return r.json(); }).then(afficherEtat).catch(function () {});
            }
          }
        })
        .catch(function () {
          if (bouton) { bouton.disabled = false; bouton.textContent = 'Je r\u00e9serve ma place'; }
          erreur('Connexion impossible. V\u00e9rifie ta connexion et r\u00e9essaie.');
        });
    });
  })();

  }
  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', usnInit);
  } else {
    usnInit();
  }
})();

/* Tout le code attend que la page soit construite. Sans cela, si le script
   est insere avant le contenu (ce que font certains plugins WordPress), il
   ne trouve aucun element et s'arrete sans rien faire. */
(function () {
  /* ---------------------------------------------------------------
     POURQUOI AUCUNE ESPERLUETTE DANS CE FICHIER

     WordPress retouche la typographie de tout ce qu'on lui confie. Sur
     du texte c'est utile ; sur du code c'est destructeur : il remplace
     l'esperluette par son entite HTML, et le navigateur s'arrete alors
     sur une erreur de syntaxe. Tout le script meurt, pas seulement la
     ligne. Constate en ligne le 20 octobre 2026 dans le bloc HTML de la
     page, puis de nouveau le 26 dans l'onglet JavaScript de
     l'extension : aucun des deux endroits n'est a l'abri.

     L'operateur " et " logique, qui s'ecrit avec deux esperluettes, est
     donc partout remplace par un ternaire " a ? b : false ". Il dit la
     meme chose et s'arrete de la meme facon quand a est faux, ce qui
     protege les acces du genre " a ? a.b : null ".

     Si tu ajoutes du code ici, n'y mets aucune esperluette : le
     generateur refuse de produire les fichiers tant qu'il en reste une.
     --------------------------------------------------------------- */
  function usnInit() {

    /* Le script est injecte par l'extension " CSS et JS personnalises ",
       donc sur TOUTES les pages du site. Celui-ci ne concerne que la
       page " A venir " : s'il ne la trouve pas, il s'arrete ici et ne
       touche a rien. */
    if (!document.getElementById('usn-agenda')) return;

  /* ---- Compte a rebours ---- */
  (function () {
    var box = document.getElementById('usn-countdown');
    if (!box) return;
    var target = new Date(box.getAttribute('data-target') || '').getTime();
    if (isNaN(target)) return;
    var els = {
      days: box.querySelector('[data-cd="days"]'),
      hours: box.querySelector('[data-cd="hours"]'),
      mins: box.querySelector('[data-cd="mins"]'),
      secs: box.querySelector('[data-cd="secs"]')
    };
    function pad(n) { return (n < 10 ? '0' : '') + n; }
    function tick() {
      var diff = Math.max(0, target - Date.now());
      var s = Math.floor(diff / 1000);
      var d = Math.floor(s / 86400); s -= d * 86400;
      var h = Math.floor(s / 3600); s -= h * 3600;
      var m = Math.floor(s / 60); s -= m * 60;
      if (els.days) els.days.textContent = d;
      if (els.hours) els.hours.textContent = pad(h);
      if (els.mins) els.mins.textContent = pad(m);
      if (els.secs) els.secs.textContent = pad(s);
    }
    tick();
    setInterval(tick, 1000);
  })();


  /* ---- Le son du film de la carte ----
     La lecture automatique n'est autorisee que sans son : le film part
     donc muet et ce bouton rend la main au visiteur. Pas de retour au
     debut a l'activation - la musique appartient au film, elle est donc
     deja en place, ou qu'on en soit dans la boucle. */
  (function () {
    var video = document.querySelector('#usn-agenda .usn-hero-film video');
    var bouton = document.getElementById('usn-son');
    if (!video || !bouton) return;

    function aDuSon() {
      // Chaque navigateur renseigne un indicateur different ; aucun n'est
      // disponible partout, d'ou les trois essais successifs.
      if (typeof video.mozHasAudio === 'boolean') return video.mozHasAudio;
      if (video.webkitAudioDecodedByteCount !== undefined) {
        return video.webkitAudioDecodedByteCount > 0;
      }
      if (video.audioTracks) return video.audioTracks.length > 0;
      return true;   // dans le doute, on laisse le bouton
    }

    function verifier() {
      // Pas de piste audio dans le fichier servi : le bouton disparait
      // plutot que de promettre un son qui n'existe pas.
      if (!aDuSon()) {
        var ligne = bouton.parentNode;
        if (ligne) ligne.parentNode.removeChild(ligne); else bouton.remove();
      }
    }
    if (video.readyState >= 2) { verifier(); }
    video.addEventListener('loadeddata', verifier);

    function libelle() {
      var texte = bouton.querySelector('.usn-son-texte');
      var icone = bouton.querySelector('.usn-son-icone');
      if (texte) texte.textContent = video.muted ? 'Activer le son' : 'Couper le son';
      if (icone) icone.textContent = video.muted ? '\u266A' : '\u266B';
      bouton.setAttribute('aria-pressed', video.muted ? 'false' : 'true');
    }
    libelle();

    bouton.addEventListener('click', function () {
      video.muted = !video.muted;
      if (!video.muted) { video.volume = 0.8; video.play(); }
      libelle();
    });
  })();


  /* =================================================================
     LA ROTATION DES PROCHAINS RENDEZ-VOUS

     Chaque vue reste a l'ecran le temps que dit son attribut
     data-usn-duree. L'avancement n'est pas compte par une minuterie mais
     porte par l'animation de la barre : c'est elle qui, en arrivant au
     bout, appelle la ville suivante. Deux consequences utiles :
       - la barre et le changement de vue ne peuvent pas se desynchroniser,
       - mettre l'animation en pause (au survol) suspend tout, puisque
         l'animation ne finit plus.
     ================================================================= */
  (function () {
    var boite = document.getElementById('usn-tourne');
    if (!boite) return;
    var vues   = boite.querySelectorAll('.usn-vue');
    var barres = boite.querySelectorAll('.usn-tourne-barre');
    if (!vues.length || vues.length !== barres.length) return;

    var reduit = window.matchMedia ? window.matchMedia('(prefers-reduced-motion: reduce)').matches : false;
    var actif = 0, enPause = false;

    /* La largeur des barres suit la duree des vues : la ville qui reste
       le plus longtemps a l'ecran occupe la plus large. */
    var durees = [], mini = Infinity, i;
    for (i = 0; i < vues.length; i++) {
      var d = parseInt(vues[i].getAttribute('data-usn-duree'), 10) || 4200;
      durees.push(d);
      if (d < mini) mini = d;
    }
    for (i = 0; i < barres.length; i++) {
      barres[i].style.flexGrow = (durees[i] / mini).toFixed(3);
    }

    function remplissage(n) { return barres[n].querySelector('i b'); }

    function montre(n) {
      actif = (n + vues.length) % vues.length;

      for (var k = 0; k < vues.length; k++) {
        var est = (k === actif);
        vues[k].classList.toggle('est-active', est);
        barres[k].classList.toggle('est-active', est);
        barres[k].setAttribute('aria-current', est ? 'true' : 'false');

        var b = remplissage(k);
        if (b ? !est : false) { b.style.animation = 'none'; b.style.width = ''; }
      }

      var plein = remplissage(actif);
      if (plein ? !reduit : false) {
        // Remise a zero forcee : sans la lecture de offsetWidth, le
        // navigateur reutilise l'animation en cours au lieu de la relancer.
        plein.style.animation = 'none';
        void plein.offsetWidth;
        plein.style.animation = 'usn-remplit ' + durees[actif] + 'ms linear forwards';
        // La forme courte " animation " vient de remettre la lecture en
        // marche : si le visiteur avait la souris sur le bloc, il faut la
        // remettre en pause tout de suite.
        appliquePause();
      }
    }

    /* La pause se pose sur l'animation elle-meme, et non par une regle de
       la feuille de style : le style en ligne pose ci-dessus l'emporterait
       sur elle. */
    function appliquePause() {
      var plein = remplissage(actif);
      if (plein) plein.style.animationPlayState = enPause ? 'paused' : 'running';
    }

    /* C'est la fin de l'animation de la barre qui fait tourner la page. */
    boite.addEventListener('animationend', function (e) {
      if (e.animationName !== 'usn-remplit') return;
      if (e.target !== remplissage(actif)) return;
      montre(actif + 1);
    });

    for (i = 0; i < barres.length; i++) {
      barres[i].addEventListener('click', function () {
        montre(parseInt(this.getAttribute('data-usn-va'), 10) || 0);
      });
    }
    var fleches = boite.querySelectorAll('.usn-tourne-f');
    for (i = 0; i < fleches.length; i++) {
      fleches[i].addEventListener('click', function () {
        montre(actif + (parseInt(this.getAttribute('data-usn-pas'), 10) || 1));
      });
    }

    /* Pause a la lecture : au survol, au doigt pose, et au clavier. */
    function pause(oui) { enPause = oui; appliquePause(); }
    boite.addEventListener('mouseenter', function () { pause(true); });
    boite.addEventListener('mouseleave', function () { pause(false); });
    boite.addEventListener('focusin',    function () { pause(true); });
    boite.addEventListener('focusout',   function () { pause(false); });
    boite.addEventListener('touchstart', function () { pause(true); }, { passive: true });
    boite.addEventListener('touchend',   function () { pause(false); }, { passive: true });

    /* La rotation demarre sur Tarbes : la vue qui reste le plus longtemps
       a l'ecran est celle que le visiteur doit voir en arrivant. */
    var depart = 0;
    for (i = 0; i < durees.length; i++) if (durees[i] > durees[depart]) depart = i;
    montre(depart);
  })();


  /* =================================================================
     MOTEUR D'ANIMATION AU SCROLL
     Reproduit le comportement de GSAP ScrollTrigger en mode " scrub " :
     - les elements partent d'une demi-hauteur d'ecran plus bas (50svh),
     - ils remontent AU RYTHME DU SCROLL (et redescendent si on remonte),
     - avec un chevauchement entre les elements d'un meme groupe,
     - le tout lisse (inertie) pour un rendu fluide.
     Sans JS ou avec " prefers-reduced-motion ", tout reste visible.
     ================================================================= */
  (function () {
    var root = document.getElementById('usn-agenda');
    if (!root) return;
    var reduce = window.matchMedia ? window.matchMedia('(prefers-reduced-motion: reduce)').matches : false;

    function clamp01(v) { return v < 0 ? 0 : v > 1 ? 1 : v; }
    function easeOut(t) { return 1 - Math.pow(1 - t, 3); }
    function vh() { return window.innerHeight || 800; }

    /* ---------- 1. Decoupe la punchline en mots ---------- */
    var punch = document.getElementById('usn-punch');
    var words = [];
    if (punch) {
      var lines = punch.querySelectorAll('[data-usn-reveal]');
      for (var l = 0; l < lines.length; l++) {
        var parts = (lines[l].textContent || '').trim().split(/\s+/);
        lines[l].textContent = '';
        for (var k = 0; k < parts.length; k++) {
          var sp = document.createElement('span');
          sp.className = 'usn-word';
          sp.textContent = parts[k];
          lines[l].appendChild(sp);
          if (k < parts.length - 1) lines[l].appendChild(document.createTextNode(' '));
          words.push(sp);
        }
      }
      for (var w = 0; w < words.length; w++) {
        var a = Math.sin(w * 12.9898) * 43758.5453; a -= Math.floor(a);
        var b = Math.sin(w * 78.233) * 12543.123;  b -= Math.floor(b);
        words[w].__dx = (a - 0.5) * 2 * Math.min(window.innerWidth * 0.45, 520);
        words[w].__dy = (b - 0.5) * 2 * 260;
        words[w].__rot = (a - 0.5) * 55;
      }
    }

    /* ---------- 1 bis. Decoupe le titre en lettres ----------
       " A venir " se leve lettre apres lettre quand on arrive dessus.
       Le decoupage se fait ici, et non dans le HTML : ecrire trente
       balises a la main rendrait le titre illisible a la relecture, et
       un lecteur d'ecran lirait les lettres une a une. La balise garde
       donc son mot entier dans aria-label. */
    (function () {
      var titre = root.querySelector('.usn-hero h1');
      if (!titre) return;
      var mot = (titre.textContent || '').trim();
      titre.setAttribute('aria-label', mot);
      titre.textContent = '';
      for (var i = 0; i < mot.length; i++) {
        var l = document.createElement('span');
        l.className = 'usn-glyphe';
        l.setAttribute('aria-hidden', 'true');
        l.textContent = mot.charAt(i);
        titre.appendChild(l);
      }
    })();

    if (reduce) return; // mouvement desactive : tout reste en place

    /* ---------- 2. Declare les groupes animes ---------- */
    // rise : monte depuis 50svh ; slide : arrive de la gauche ; pop : grossit
    var groups = [];
    function group(sel, opts) {
      var els = root.querySelectorAll(sel);
      if (els.length) groups.push({ els: els, o: opts });
    }
    // Les lettres du titre d'abord : elles doivent se lever avant tout
    // le reste, c'est la premiere chose que le visiteur atteint.
    group('.usn-glyphe',  { type: 'lettre', dist: 0, stagger: 0.075, span: 0.40 });
    group('.usn-an',      { type: 'rise',  dist: 0.22, stagger: 0,    span: 0.45 });
    group('.usn-ligne',   { type: 'slide', dist: 0.34, stagger: 0.13, span: 0.58 });
    group('.usn-capte-carte', { type: 'rise', dist: 0.18, stagger: 0, span: 0.45 });
    group('.usn-affiche', { type: 'rise',  dist: 0.40, stagger: 0.11, span: 0.58 });
    group('.usn-mot',     { type: 'rise',  dist: 0.26, stagger: 0.14, span: 0.50 });

    var all = [];
    for (var g = 0; g < groups.length; g++) {
      for (var i = 0; i < groups[g].els.length; i++) {
        var el = groups[g].els[i];
        all.push({
          el: el, i: i, o: groups[g].o, p: 0, top: 0,
          rot: parseFloat(el.getAttribute('data-rot') || '0')
        });
      }
    }

    function scrollY() { return window.pageYOffset || document.documentElement.scrollTop || 0; }

    /* Position NATURELLE de chaque element (transform neutralise), mise en
       cache : sinon le decalage applique fausse la mesure et l'element
       n'entre jamais dans la zone de declenchement. */
    function measureAll() {
      for (var i = 0; i < all.length; i++) {
        var it = all[i], prev = it.el.style.transform;
        it.el.style.transform = 'none';
        it.top = it.el.getBoundingClientRect().top + scrollY();
        it.el.style.transform = prev;
      }
    }
    measureAll();

    /* ---------- Portraits : couleur au passage au centre ----------
       Chaque photo se colore a mesure qu'elle approche du centre de
       l'ecran, puis retourne au noir et blanc en s'en eloignant. */
    var photos = [];
    (function () {
      var imgs = root.querySelectorAll('.usn-affiche img');
      for (var i = 0; i < imgs.length; i++) photos.push({ el: imgs[i], p: 0 });
    })();

    function cibleCouleur(ph) {
      var r = ph.el.getBoundingClientRect();
      if (!r.height) return 0;
      var h = vh();
      var ecart = Math.abs((r.top + r.height / 2) - h / 2);
      // Pleine couleur au centre, noir et blanc au-dela de 42% de l'ecran.
      return clamp01(1 - ecart / (h * 0.42));
    }

    function peindre(ph) {
      var e = easeOut(ph.p);
      ph.el.style.filter = 'grayscale(' + (1 - e).toFixed(3) + ') contrast(' + (1 + 0.06 * (1 - e)).toFixed(3) + ')';
      ph.el.style.transform = 'scale(' + (1 + 0.035 * e).toFixed(4) + ')';
    }
    for (var ip = 0; ip < photos.length; ip++) peindre(photos[ip]);

    // etat initial immediat (evite tout " flash " de contenu deja en place)
    for (var s = 0; s < all.length; s++) apply(all[s], 0);

    /* ---------- 3. Calcul de l'avancement lie au scroll ---------- */
    function target(item) {
      var h = vh();
      var top = item.top - scrollY();
      // 0 quand le haut de l'element entre par le bas, 1 quand il a monte
      var raw = (h * 0.92 - top) / (h * item.o.span);
      var st = item.i * item.o.stagger;
      return clamp01((clamp01(raw) - st) / Math.max(0.2, 1 - st));
    }

    function apply(item, p) {
      var e = easeOut(p), inv = 1 - e, o = item.o, st = item.el.style;
      var rot = item.rot ? ' rotate(' + item.rot + 'deg)' : '';
      if (o.type === 'lettre') {
        // En pourcentage de la lettre elle-meme, et non en pixels : une
        // lettre doit se lever de sa propre hauteur, quelle que soit la
        // taille a laquelle le titre s'affiche.
        st.transform = 'translate3d(0,' + (inv * 108).toFixed(1) + '%,0)';
      } else if (o.type === 'rise') {
        st.transform = 'translate3d(0,' + (inv * o.dist * vh()).toFixed(1) + 'px,0)' + rot;
      } else if (o.type === 'slide') {
        st.transform = 'translate3d(' + (-inv * o.dist * Math.min(window.innerWidth, 900)).toFixed(1) + 'px,0,0)' + rot;
      } else if (o.type === 'slideR') {
        st.transform = 'translate3d(' + (inv * o.dist * Math.min(window.innerWidth, 900)).toFixed(1) + 'px,0,0)' + rot;
      } else { // pop
        st.transform = 'scale(' + (0.3 + 0.7 * e).toFixed(3) + ')' + rot;
      }
      // Les lettres ne s'estompent pas : elles sortent de derriere le bord
      // du titre. Une lettre a demi transparente a demi sortie ferait sale.
      if (o.type !== 'lettre') st.opacity = e.toFixed(3);
    }

    /* ---------- 4. Punchline : les mots se rassemblent ---------- */
    var punchP = 0;
    // Ancre sur la premiere ligne de texte (et non sur la section, bien plus
    // haute) : les mots se rassemblent des que le texte entre a l'ecran.
    var punchAnchor = punch ? punch.querySelector('[data-usn-reveal]') : null;
    function punchTarget() {
      if (!punchAnchor) return 0;
      var r = punchAnchor.getBoundingClientRect(), h = vh();
      return clamp01((h * 1.0 - r.top) / (h * 0.5));
    }
    function punchApply(p) {
      var n = words.length;
      for (var i = 0; i < n; i++) {
        var t = clamp01((p - (i / n) * 0.45) / 0.55);
        var e = easeOut(t), inv = 1 - e, ws = words[i];
        ws.style.transform = 'translate3d(' + (ws.__dx * inv).toFixed(1) + 'px,' + (ws.__dy * inv).toFixed(1) + 'px,0) rotate(' + (ws.__rot * inv).toFixed(1) + 'deg)';
        ws.style.opacity = e.toFixed(3);
        ws.style.filter = inv > 0.02 ? 'blur(' + (inv * 9).toFixed(1) + 'px)' : 'none';
      }
      var sub = punch ? punch.querySelector('[data-usn-sub]') : null;
      if (sub) {
        var sp = clamp01((p - 0.6) / 0.4);
        sub.style.opacity = sp.toFixed(3);
        sub.style.transform = 'translate3d(0,' + ((1 - sp) * 40).toFixed(1) + 'px,0)';
      }
    }
    punchApply(0);

    /* ---------- 5. Boucle de rendu (lissage type ScrollSmoother) ---------- */
    var running = false;
    function frame() {
      var moving = false;

      for (var i = 0; i < all.length; i++) {
        var it = all[i], t = target(it);
        it.p += (t - it.p) * 0.12;             // lissage
        if (Math.abs(t - it.p) > 0.001) moving = true; else it.p = t;
        apply(it, it.p);
      }

      if (punch) {
        var tp = punchTarget();
        punchP += (tp - punchP) * 0.12;
        if (Math.abs(tp - punchP) > 0.001) moving = true; else punchP = tp;
        punchApply(punchP);
      }

      for (var k = 0; k < photos.length; k++) {
        var ph = photos[k], tc = cibleCouleur(ph);
        ph.p += (tc - ph.p) * 0.14;
        if (Math.abs(tc - ph.p) > 0.004) moving = true; else ph.p = tc;
        peindre(ph);
      }

      if (moving) { requestAnimationFrame(frame); } else { running = false; }
    }
    function kick() { if (!running) { running = true; requestAnimationFrame(frame); } }

    window.addEventListener('scroll', kick, { passive: true });
    window.addEventListener('resize', function () { measureAll(); kick(); });
    window.addEventListener('load', function () { measureAll(); kick(); });
    kick();
  })();


  /* ---- Les formulaires " rester prevenu " ----
     La page en porte deux - un sous la rotation des villes, un en bas -
     qui vont vers la meme liste Brevo. Un seul bout de code les sert :
     chacun se designe par data-usn-brevo, et dit dans data-usn-msg ou
     ecrire son accuse de reception.

     Pas de compteur de places ici : la page annonce des dates, elle ne
     reserve rien. L'envoi part vers Brevo dans un cadre cache. */
  (function () {
    var formes = document.querySelectorAll('#usn-agenda form[data-usn-brevo]');

    function brancher(form) {
      var msg = document.getElementById(form.getAttribute('data-usn-msg') || '');

      form.addEventListener('submit', function (e) {
        var nom   = form.querySelector('[data-usn-nom]');
        var email = form.querySelector('[data-usn-email]');
        if (!nom || !email) return;
        if (!nom.value.trim() || !/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(email.value)) {
          e.preventDefault();
          (nom.value.trim() ? email : nom).focus();
          return;
        }
        if (msg) {
          msg.textContent = 'Merci ! Vous serez pr\u00e9venu d\u00e8s qu\'une nouvelle date est annonc\u00e9e.';
          msg.style.display = 'block';
        }
        setTimeout(function () { form.reset(); }, 100);
      });
    }

    for (var i = 0; i < formes.length; i++) brancher(formes[i]);
  })();

  }
  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', usnInit);
  } else {
    usnInit();
  }
})();
